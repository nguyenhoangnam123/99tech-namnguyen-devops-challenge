$ curl -s -o /dev/null -w "%{http_code}" localhost:8080<path>
/           -> 200
/api/users  -> 502

$ docker compose logs nginx | grep upstream

{"ok":true,"time":{"now":"2026-11-18T04:27:22.341Z"}}

$ docker compose exec api wget -qO- localhost:3001/api/users
